getConstraints() =>
  let list = [];
  if (currentSetting !== "none") list.push(`Genre/Setting: ${worldSettings[currentSetting].description}`);
  // Tone — multi-select
  if (currentTones && currentTones.length > 0) {
    let toneDescs = currentTones.map(t => tones[t] ? tones[t].description : t).join(" + ");
    list.push(`Tone: ${toneDescs}`);
  } else if (currentTone !== "none") {
    list.push(`Tone: ${tones[currentTone].description}`);
  }
  // Archetype — multi-select
  if (currentArchetypes && currentArchetypes.length > 0) {
    let archDescs = currentArchetypes.map(a => archetypes[a] ? archetypes[a].description : a).join(" + ");
    list.push(`Character Archetype: ${archDescs}`);
  } else if (currentArchetype !== "none") {
    list.push(`Character Archetype: ${archetypes[currentArchetype].description}`);
  }
  // Dynamics — multi-select
  if (currentDynamicsList && currentDynamicsList.length > 0) {
    let dynDescs = currentDynamicsList.map(d => dynamics[d] ? dynamics[d].description : d).join(" + ");
    list.push(`Dynamics: ${dynDescs}`);
  } else if (currentDynamics !== "none") {
    list.push(`Dynamics: ${dynamics[currentDynamics].description}`);
  }
  if (currentCharacterCount !== "none") list.push(`Characters: ${characterCounts[currentCharacterCount].description}`);
  if (currentGender !== "none") list.push(`Main Character Gender: ${genders[currentGender].description}`);
  let name = userNameEl.value.trim();
  if (name !== "") {
    list.push(`Include a character named "${literal(name)}" as the protagonist/main character.`);
  }
  if (list.length > 0) {
    return "\nUse the following constraints/parameters for the generation:\n" + list.map(item => "- " + item).join("\n");
  }
  return "";
