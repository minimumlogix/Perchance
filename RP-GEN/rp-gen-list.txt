getConstraints() =>
  let list = [];
  if (currentSetting !== "none") list.push(`Genre/Setting: ${worldSettings[currentSetting].description}`);
  if (currentTone !== "none") list.push(`Tone: ${tones[currentTone].description}`);
  if (currentArchetype !== "none") list.push(`Character Archetype: ${archetypes[currentArchetype].description}`);
  if (currentDynamics !== "none") list.push(`Dynamics: ${dynamics[currentDynamics].description}`);
  if (currentCharacterCount !== "none") list.push(`Characters: ${characterCounts[currentCharacterCount].description}`);
  if (currentGender !== "none") list.push(`Main Character Gender: ${genders[currentGender].description}`);
  let name = userNameEl.value.trim();
  if (name !== "") {
    list.push(`Include a character named "${literal(name)}" as the protagonist/main character.`);
  }
  if (list.length > 0) {
    return "\nUse the following constraints/parameters for the generation:\n" + list.map(item => "- " + item).join("\n");
  }
  return "";
