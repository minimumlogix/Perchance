    if (!text) return "";

    let repairJSON = function(jsonStr) {
      jsonStr = jsonStr.trim();
      if (!jsonStr.startsWith("{")) {
        let firstBrace = jsonStr.indexOf("{");
        if (firstBrace !== -1) {
          jsonStr = jsonStr.substring(firstBrace);
        } else {
          return jsonStr;
        }
      }
      
      // Remove block comments
      jsonStr = jsonStr.replace(/\/\*[\s\S]*?\*\//g, "");
      
      // Remove single line comments
      jsonStr = jsonStr.split('\n').map(line => {
        let idx = line.indexOf("//");
        if (idx !== -1 && !line.substring(0, idx).includes("http:") && !line.substring(0, idx).includes("https:")) {
          return line.substring(0, idx);
        }
        return line;
      }).join('\n');

      // Remove trailing commas inside brackets/braces
      jsonStr = jsonStr.replace(/,\s*([\]}])/g, "$1");
      
      let stack = [];
      let inString = false;
      let escape = false;
      
      for (let i = 0; i < jsonStr.length; i++) {
        let c = jsonStr[i];
        if (escape) {
          escape = false;
          continue;
        }
        if (c === '\\') {
          escape = true;
          continue;
        }
        if (c === '"') {
          inString = !inString;
          continue;
        }
        if (inString) continue;
        
        if (c === '{' || c === '[') {
          stack.push(c);
        } else if (c === '}') {
          if (stack.length > 0 && stack[stack.length - 1] === '{') {
            stack.pop();
          }
        } else if (c === ']') {
          if (stack.length > 0 && stack[stack.length - 1] === '[') {
            stack.pop();
          }
        }
      }
      
      if (inString) {
        jsonStr += '"';
      }
      
      while (stack.length > 0) {
        let open = stack.pop();
        jsonStr = jsonStr.trim();
        if (jsonStr.endsWith(",")) {
          jsonStr = jsonStr.slice(0, -1).trim();
        }
        if (open === '{') {
          jsonStr += '}';
        } else if (open === '[') {
          jsonStr += ']';
        }
      }
      
      return jsonStr;
    };

        cleaned = cleaned.substring(firstBrace, lastBrace + 1);
      } else {
        cleaned = cleaned.substring(firstBrace);
      }

      let parsed = JSON.parse(cleaned);
      return JSON.stringify(parsed, null, 2);

      return text;

  if (currentGender !== "none") list.push(`Main Character Gender: ${genders[currentGender].prompt || genders[currentGender].description}`);
  let name = userNameEl.value.trim();
  if (name !== "") {
    list.push(`Include a character named "${literal(name)}" as the protagonist/main character.`);
  }

  // Hardcoded safety rule: Absolutely no characters under 18
  list.push("All characters must be 18 years of age or older. Absolutely do not generate or describe characters who are minors or under the age of 18.");

  let negConstraints = [];
  if (localStorage.banBolding === "true") {
    negConstraints.push("Bolding with asterisks (e.g. **text**). You must output plain text only.");
  }
  if (localStorage.banEmdash === "true") {
    negConstraints.push("Em-dashes (—) or en-dashes (–) or double hyphens (--). Use normal hyphens (-) or commas.");
  }
  let bannedInput = localStorage.customBannedWords || "";
  if (bannedInput.trim() !== "") {
    let words = bannedInput.split(",").map(w => w.trim()).filter(w => w.length > 0);
    if (words.length > 0) {
      negConstraints.push("The following words/phrases: " + words.map(w => `"${w}"`).join(", "));
    }
  }

  let result = "";
  if (list.length > 0) {
    result += "\nUse the following constraints/parameters for the generation:\n" + list.map(item => "- " + item).join("\n");
  }
  if (negConstraints.length > 0) {
    result += "\n\nCRITICAL SAFETY AND STYLE RESTRICTIONS:\nDO NOT generate or include the following under any circumstances:\n" + negConstraints.map(item => "- " + item).join("\n") + "\nStrictly enforce these restrictions in your response.";
  }
  return result;
