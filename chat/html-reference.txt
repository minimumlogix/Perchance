    async function seedDefaultDataIfNeeded() {

      if (count === 0) {
        for (const char of DEFAULT_CHARACTERS) {
          await db.characters.put(char);
          
          // Create initial thread
          const threadId = "thread_" + char.id;
          await db.threads.put({
            id: threadId,
            characterId: char.id,
            lastUpdated: Date.now()
          });

          await db.messages.put({
            id: "msg_greeting_" + char.id,
            threadId: threadId,
            characterId: char.id,

            content: char.greeting,

    async function renderMessages() {
      const container = document.getElementById("messagesContainer");
      // Keep narrative card, remove existing message rows
      container.querySelectorAll(".message-row").forEach(el => el.remove());

      if (!activeThread) return;

      const messages = await db.messages.where({ threadId: activeThread.id }).sortBy("timestamp");

      for (const msg of messages) {
        const isUser = msg.role === "user";
        const row = document.createElement("div");
        row.className = `message-row ${isUser ? 'user-row' : 'ai-row'}`;
        row.dataset.msgId = msg.id;

        const avatarSrc = isUser ? "assets/lysandra.png" : activeCharacter.avatarUrl;
        const authorName = isUser ? "You" : activeCharacter.name;
        const formattedTime = new Date(msg.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

        // Parse markdown & italics
        let parsedContent = marked.parse(msg.content);
        parsedContent = DOMPurify.sanitize(parsedContent);

        let imageHtml = msg.imageUrl ? `<img src="${msg.imageUrl}" alt="Generated Illustration">` : '';

        row.innerHTML = `
          <img src="${avatarSrc}" class="message-avatar" alt="${authorName}">
          <div class="message-content-wrapper">
            <div class="message-meta">
              <span class="message-author">${authorName}</span>
              <span>${formattedTime}</span>
            </div>

              <button class="action-sub-btn" onclick="deleteMessage('${msg.id}')">🗑 Delete</button>
            </div>
          </div>
        `;

        container.appendChild(row);
      }

    }

        id: userMsgId,
        threadId: activeThread.id,
        characterId: activeCharacter.id,
        role: "user",
        content: text,
        timestamp: Date.now()
      });

      await renderMessages();

      // Simulate AI Character Streaming Response
      await generateAIResponse();

      }

    function speakText(text) {
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const cleanText = text.replace(/\*.*?\*/g, ''); // strip action brackets
        const utterance = new SpeechSynthesisUtterance(cleanText);
        utterance.rate = 0.95;
        utterance.pitch = 1.0;

        const waveform = document.getElementById("waveform");
        const btnPlay = document.getElementById("btnPlayVoice");

        utterance.onstart = () => {
          waveform.classList.add("playing");
          btnPlay.textContent = "⏸";
        };
        utterance.onend = () => {
          waveform.classList.remove("playing");
          btnPlay.textContent = "▶";
        };
        utterance.onerror = () => {
          waveform.classList.remove("playing");
          btnPlay.textContent = "▶";
        };

        window.speechSynthesis.speak(utterance);
      } else {
        alert("Text-to-speech is not supported in this browser.");
      }
    }

    function escapeHtml(str) {
      if (!str) return '';
      return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
    }

    function escapeJsString(str) {
      if (!str) return '';
      return str.replace(/\\/g, '\\\\').replace(/`/g, '\\`').replace(/\${/g, '\\${');
    }

    function formatTimestamp(ts) {
      const diff = Math.floor((Date.now() - ts) / 1000);
      if (diff < 60) return "Just now";
      if (diff < 3600) return `${Math.floor(diff / 60)}m ago`;
      if (diff < 86400) return `${Math.floor(diff / 3600)}h ago`;
      return `${Math.floor(diff / 86400)}d ago`;
    }

    function delay(ms) {
      return new Promise(resolve => setTimeout(resolve, ms));
    }
